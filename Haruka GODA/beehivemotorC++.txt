//このコードを動かすには、事前にArduino IDEのライブラリ管理画面で「Adafruit PWM Servo Driver Library」を検索してインストールしておいてください。

C++
#include <Wire.h>
#include <Adafruit_PWMServoDriver.h>

// PCA9685の初期化（デフォルトのI2Cアドレス 0x40）
Adafruit_PWMServoDriver pwm = Adafruit_PWMServoDriver();

// サーボモーターのパルス幅の設定（ここが最重要！）
// 50Hz(周期20ms)における、0度と180度のときのカウント値(0-4095)を指定します。
#define SERVOMIN  150 // 0度のときのパルス幅（目安）
#define SERVOMAX  500 // 180度のときのパルス幅（目安）

// サーボモーターを接続したPCA9685のピン番号（0番〜15番のどこか）
uint8_t servoNum = 0; 

void setup() {
  Serial.begin(9600);
  Serial.println("PCA9685 Servo Test Start!");

  pwm.begin();
  pwm.setOscillatorFrequency(27000000);
  pwm.setPWMFreq(50);  // アナログサーボは基本50Hzで駆動します

  delay(10);
}

// 角度（0〜180）をPCA9685用のパルス値（SERVOMIN〜SERVOMAX）に変換する関数
int angleToPulse(int angle) {
  return map(angle, 0, 180, SERVOMIN, SERVOMAX);
}

void loop() {
  // 1. 0度から180度までゆっくり回す
  Serial.println("Moving to 180 degrees...");
  for (int angle = 0; angle <= 180; angle += 10) {
    pwm.setPWM(servoNum, 0, angleToPulse(angle));
    delay(30); // 動くスピードの調整
  }
  delay(1000); // 1秒キープ

  // 2. 180度から0度までゆっくり戻す
  Serial.println("Moving to 0 degrees...");
  for (int angle = 180; angle >= 0; angle -= 10) {
    pwm.setPWM(servoNum, 0, angleToPulse(angle));
    delay(30);
  }
  delay(1000); // 1秒キープ
}
